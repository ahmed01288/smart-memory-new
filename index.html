افتح لوحة تحكم Supabase ⬅️ ادخل على SQL Editor ⬅️ الصق الكود التالي واضغط Run: 

create table if not exists contacts (
  id text primary key,
  text text,
  person_name text,
  category text,
  phone text,
  created_at text,
  updated_at text
);
alter table contacts enable row level security;
create policy "Public Access" on contacts for all using (true) with check (true);
